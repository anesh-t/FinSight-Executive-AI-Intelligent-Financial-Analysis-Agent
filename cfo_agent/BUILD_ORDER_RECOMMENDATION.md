```python
class MasterOrchestrator:
    def __init__(self):
        self.rag_agent = RAGAgent()
        self.sql_agent = SQLAgent()
        self.classifier = QueryClassifier()
    
    def query(self, question: str):
        # Classify intent
        intent = self.classifier.classify(question)
        
        # Route to appropriate agent(s)
        if intent == 'QUALITATIVE':
            return self.rag_agent.query(question)
        
        elif intent == 'QUANTITATIVE':
            return self.sql_agent.query(question)
        
        elif intent == 'HYBRID':
            # Run both in parallel
            rag_result = self.rag_agent.query(question)
            sql_result = self.sql_agent.query(question)
            
            # Synthesize
            return self.synthesize(rag_result, sql_result)
```
